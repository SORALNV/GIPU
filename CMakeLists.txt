cmake_minimum_required(VERSION 3.24)
project(GIPU VERSION 0.1.0 LANGUAGES CXX)
option(GIPU_ENABLE_GPU "nvCOMP GPUバックエンドを組み込む" OFF)
option(GIPU_ENABLE_LIBDEFLATE "検出できればlibdeflateのCPU経路を組み込む" ON)
option(GIPU_ENABLE_ISAL "検出できればISA-LのCPU経路を組み込む" ON)
option(GIPU_SANITIZE "CPUコードのASan/UBSanを有効にする" OFF)
option(GIPU_GPU_TESTS "実機GPUの統合テストをCTestに登録する" OFF)
option(GIPU_ENABLE_RAPIDGZIP "実験用のRapidgzip単一ストリーム並列CPU経路" OFF)
set(RAPIDGZIP_ROOT "" CACHE PATH "Rapidgzip/librapidarchiveのソースディレクトリ")
set(NVCOMP_ROOT "" CACHE PATH "nvCOMP SDKのディレクトリ")
set(CUDA_ROOT "" CACHE PATH "CUDAランタイムのディレクトリ")
find_package(ZLIB REQUIRED)
add_library(gipu_core src/zip.cpp src/io.cpp src/cpu.cpp src/libdeflate.cpp src/checksum.cpp src/rapidgzip.cpp src/backend.cpp src/hybrid.cpp)
target_include_directories(gipu_core PUBLIC include)
target_compile_features(gipu_core PUBLIC cxx_std_20)
target_compile_options(gipu_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_link_libraries(gipu_core PUBLIC ZLIB::ZLIB)
find_package(Threads REQUIRED)
target_link_libraries(gipu_core PUBLIC Threads::Threads)
if(GIPU_ENABLE_RAPIDGZIP)
  find_path(RAPIDGZIP_INCLUDE rapidgzip/ParallelGzipReader.hpp HINTS "${RAPIDGZIP_ROOT}/src" REQUIRED)
  target_include_directories(gipu_core SYSTEM PRIVATE "${RAPIDGZIP_INCLUDE}")
  target_compile_definitions(gipu_core PRIVATE GIPU_HAVE_RAPIDGZIP=1)
endif()
if(GIPU_ENABLE_LIBDEFLATE)
  find_path(LIBDEFLATE_INCLUDE libdeflate.h)
  find_library(LIBDEFLATE_LIBRARY deflate)
endif()
if(GIPU_ENABLE_LIBDEFLATE AND LIBDEFLATE_INCLUDE AND LIBDEFLATE_LIBRARY)
  target_include_directories(gipu_core PRIVATE "${LIBDEFLATE_INCLUDE}")
  target_link_libraries(gipu_core PUBLIC "${LIBDEFLATE_LIBRARY}")
  target_compile_definitions(gipu_core PRIVATE GIPU_HAVE_LIBDEFLATE=1)
endif()
if(GIPU_ENABLE_ISAL)
  find_path(ISAL_INCLUDE igzip_lib.h PATH_SUFFIXES isa-l)
  find_library(ISAL_LIBRARY isal)
endif()
if(GIPU_ENABLE_ISAL AND ISAL_INCLUDE AND ISAL_LIBRARY)
  target_include_directories(gipu_core SYSTEM PRIVATE "${ISAL_INCLUDE}")
  target_link_libraries(gipu_core PUBLIC "${ISAL_LIBRARY}")
  target_compile_definitions(gipu_core PRIVATE GIPU_HAVE_ISAL=1)
endif()
if(GIPU_ENABLE_GPU)
  find_path(NVCOMP_INCLUDE nvcomp/native/streaming_gzip.hpp HINTS "${NVCOMP_ROOT}/include" REQUIRED)
  find_library(NVCOMP_LIBRARY nvcomp HINTS "${NVCOMP_ROOT}/lib" REQUIRED)
  find_path(CUDART_INCLUDE cuda_runtime_api.h HINTS "${CUDA_ROOT}/include" "${CUDA_ROOT}/targets/x86_64-linux/include" REQUIRED)
  find_library(CUDART_LIBRARY cudart HINTS "${CUDA_ROOT}/lib" "${CUDA_ROOT}/targets/x86_64-linux/lib" REQUIRED)
  target_sources(gipu_core PRIVATE src/gpu.cpp)
  target_include_directories(gipu_core PRIVATE "${NVCOMP_INCLUDE}" "${CUDART_INCLUDE}")
  target_link_libraries(gipu_core PUBLIC "${NVCOMP_LIBRARY}" "${CUDART_LIBRARY}")
  target_compile_definitions(gipu_core PUBLIC GIPU_ENABLE_GPU=1)
else()
  target_sources(gipu_core PRIVATE src/gpu_stub.cpp)
endif()
if(GIPU_SANITIZE)
  target_compile_options(gipu_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(gipu_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(gipu src/main.cpp)
target_link_libraries(gipu PRIVATE gipu_core)
install(TARGETS gipu RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE:gipu>")
  add_test(NAME integration_hybrid COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE:gipu>")
  set_tests_properties(integration_hybrid PROPERTIES ENVIRONMENT "GIPU_TEST_BACKEND=hybrid;GIPU_TEST_THREADS=4" RUN_SERIAL TRUE TIMEOUT 180)
  if(GIPU_ENABLE_LIBDEFLATE AND LIBDEFLATE_INCLUDE AND LIBDEFLATE_LIBRARY)
    add_test(NAME integration_libdeflate COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE:gipu>")
    set_tests_properties(integration_libdeflate PROPERTIES ENVIRONMENT "GIPU_TEST_BACKEND=libdeflate")
  endif()
  if(GIPU_ENABLE_RAPIDGZIP)
    add_test(NAME integration_rapidgzip COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE:gipu>")
    set_tests_properties(integration_rapidgzip PROPERTIES ENVIRONMENT "GIPU_TEST_BACKEND=rapidgzip;GIPU_TEST_THREADS=4")
  endif()
  if(GIPU_ENABLE_ISAL AND ISAL_INCLUDE AND ISAL_LIBRARY)
    add_test(NAME integration_isal COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE:gipu>")
    set_tests_properties(integration_isal PROPERTIES ENVIRONMENT "GIPU_TEST_BACKEND=isal")
  endif()
  if(GIPU_GPU_TESTS)
    if(NOT GIPU_ENABLE_GPU)
      message(FATAL_ERROR "GIPU_GPU_TESTSにはGIPU_ENABLE_GPUが必要です")
    endif()
    set_property(TEST integration_hybrid APPEND PROPERTY ENVIRONMENT "GIPU_EXPECT_HYBRID_GPU=1")
    foreach(mode auto stream batch)
      add_test(NAME integration_gpu_${mode} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE:gipu>")
      set_tests_properties(integration_gpu_${mode} PROPERTIES ENVIRONMENT "GIPU_TEST_BACKEND=gpu;GIPU_TEST_MODE=${mode}" RUN_SERIAL TRUE TIMEOUT 180)
    endforeach()
    add_test(NAME integration_gpu_lookahead COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE:gipu>")
    set_tests_properties(integration_gpu_lookahead PROPERTIES ENVIRONMENT "GIPU_TEST_BACKEND=gpu;GIPU_TEST_ALGORITHM=lookahead" RUN_SERIAL TRUE TIMEOUT 180)
  endif()
endif()
